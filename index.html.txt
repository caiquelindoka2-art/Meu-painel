    <script>
        // Função para abrir/fechar o menu flutuante
        function togglePanel() {
            const panel = document.getElementById('proPanel');
            panel.classList.toggle('hidden');
        }

        // Sistema de Abas
        function switchTab(evt, tabName) {
            const contents = document.getElementsByClassName('tab-content');
            for (let c of contents) {
                c.classList.remove('active');
            }

            const tabs = document.getElementsByClassName('tab-btn');
            for (let t of tabs) {
                t.classList.remove('active');
            }

            document.getElementById(tabName).classList.add('active');
            evt.currentTarget.classList.add('active');
        }

        // Seleção de Chips (Grupos)
        function selectChip(element, groupClass) {
            const parent = element.parentElement;
            const chips = parent.getElementsByClassName('chip');
            for (let c of chips) {
                c.classList.remove('selected');
            }
            element.classList.add('selected');
        }

        // Atualização de Valores na Interface
        function updateSens(val) {
            document.getElementById('sensValue').innerText = `Valor: ${val}%`;
        }

        function updateStab(val) {
            document.getElementById('stabValue').innerText = `Nível: ${val}%`;
        }

        // Simulação Visual do ESP na Arena
        function toggleEspSimulation() {
            const isChecked = document.getElementById('espToggle').checked;
            const target1 = document.getElementById('target1');
            const target2 = document.getElementById('target2');

            if (isChecked) {
                target1.classList.add('active');
                target2.classList.add('active');
            } else {
                target1.classList.remove('active');
                target2.classList.remove('active');
            }
        }

        function updateAimbotState() {
            // Feedback visual adicional opcional ao ativar o aimbot

        }
